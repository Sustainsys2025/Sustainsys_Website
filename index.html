Show me top 5 counterparties by exposure that breached VaR limits last quarter
-- generated · validated · 12 guardrails passed
SELECT c.name, SUM(p.exposure) AS total_exp,
       COUNT(b.breach_id) AS breaches
FROM counterparties c
JOIN positions p   ON p.cp_id = c.id
JOIN var_breaches b ON b.cp_id = c.id
WHERE b.quarter = '2026-Q1'
GROUP BY c.name
ORDER BY total_exp DESC
LIMIT 5;
